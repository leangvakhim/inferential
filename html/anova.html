anova_script.py
1 import pandas as pd
2 import statsmodels.api as sm
3 from statsmodels.formula.api import ols
4
5 # 1. Create the dataset
6 df = pd.DataFrame({
7     'Height': [3, 4, 5, 6, 8, 10, 5, 6, 7],
8     'Fert': ['A','A','A', 'B','B','B', 'C','C','C']
9 })
10
11 # 2. Fit the Ordinary Least Squares model
12 # Formula: Dependent ~ Categorical_Independent
13 model = ols('Height ~ C(Fert)', data=df).fit()
14
15 # 3. Generate ANOVA table
16 anova_table = sm.stats.anova_lm(model, typ=2)
17 print(anova_table)
